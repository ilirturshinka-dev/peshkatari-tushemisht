        // Filtrimi i thjeshtë i kartave të menusë
        function filterMenu(category) {
            document.querySelectorAll('.menu-filter-btn').forEach(btn => {
                if (btn.dataset.filter === category) {
                    btn.classList.remove('bg-sand', 'text-navy', 'hover:bg-sandDark');
                    btn.classList.add('bg-navy', 'text-white', 'shadow-md');
                } else {
                    btn.classList.remove('bg-navy', 'text-white', 'shadow-md');
                    btn.classList.add('bg-sand', 'text-navy', 'hover:bg-sandDark');
                }
            });

            document.querySelectorAll('.menu-card').forEach(card => {
                if (category === 'all' || card.dataset.cat === category) {
                    card.style.display = 'block';
                } else {
                    card.style.display = 'none';
                }
            });
        }

        // Ndryshimi i gjuhës (SQ <-> EN)
        function toggleLanguage() {
            currentLang = currentLang === 'sq' ? 'en' : 'sq';
            document.getElementById('langLabel').textContent = currentLang === 'sq' ? 'EN' : 'SQ';
            document.documentElement.lang = currentLang;

            document.querySelectorAll('[data-i18n]').forEach(el => {
                const key = el.getAttribute('data-i18n');
                if (translations[currentLang][key]) {
                    el.textContent = translations[currentLang][key];
                }
            });

            document.querySelectorAll('[data-sq]').forEach(el => {
                el.textContent = el.getAttribute(currentLang === 'sq' ? 'data-sq' : 'data-en');
            });
        }
